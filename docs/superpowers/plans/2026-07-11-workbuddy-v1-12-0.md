# WorkBuddy Distribution and Live Evaluation v1.12.0 Implementation Plan

> **Execution workflow:** Use superpowers:executing-plans in this session and implement the checklist sequentially. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship v1.12.0 with a formally installed WorkBuddy standard Skill, deterministic payload diagnosis, discoverable helper/frontmatter behavior, and a real WorkBuddy evaluation loop with no known P0 issue.

**Architecture:** `build.py` produces the complete standard Skill and a deterministic SHA-256 manifest. A standard-library-only doctor validates that installed payload and runtime, while the launcher forwards helper arguments verbatim and special-cases doctor so a broken runtime record remains diagnosable. Bash and PowerShell install the identical standard payload into WorkBuddy, and a recorded real-app forward test closes the loop before release.

**Tech Stack:** Python 3.11+, `tomllib`, `hashlib`, `pathlib`, `subprocess`, Bash, Windows PowerShell 5.1+, pytest, uv, GitHub Actions, macOS WorkBuddy v5.2.5.

## Global Constraints

- Release version is exactly 1.12.0.
- WorkBuddy destination is exactly `~/.workbuddy/skills/obsidian-knowledge-base`.
- The default all-platform set includes `workbuddy` in Bash and PowerShell.
- Codex, QoderWork, WorkBuddy, and the canonical support copy receive one identical complete standard payload.
- Replacing a WorkBuddy symlink removes only the link entry and never mutates its target clone.
- `manifest.json` excludes `header.md`, itself, `.DS_Store`, `__pycache__`, `.pyc`, and `.pyo`; it includes every other regular payload file, including `agents/openai.yaml`.
- Doctor is read-only, uses only the Python standard library in its own process, returns 0 only when healthy, returns 1 for an unhealthy installation, and supports `--json`.
- `run_helper.py doctor` runs with the launcher's current interpreter even when `runtime.json` is invalid; every other helper continues using the installer-selected runtime.
- Helper arguments after the first helper token are forwarded unchanged except that one compatibility `--` separator is removed.
- Frontmatter precedence is `type defaults < Vault template < stdin/content-file frontmatter < explicit CLI fields`.
- The real WorkBuddy evaluation writes one useful insight note to the real Vault; repeat only after a confirmed P0 or a fix that changes the observed live path.
- No background service, automatic repair, arbitrary destination, generic metadata flags, ordinary-note rewrite, JSON-envelope unification, audit refactor, or unrelated platform adapter is added.
- Do not tag while a P0 is known, unverified, or any required Linux/Windows gate is pending or failing.

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import hashlib
import json
import os
import subprocess
import sys
from pathlib import Path
from typing import Any

HOUSEKEEPING = {".DS_Store", "__pycache__"}
REQUIRED_RESOURCES = (
    "SKILL.md",
    "references/note-creation.md",
    "references/rules-and-errors.md",
    "assets/templates/daily-note.md",
    "scripts/run_helper.py",
)
HELPER_MODULES = (
    "audit_vault", "process_inbox", "suggest_links", "create_note",
    "update_note", "vault_info", "detect_index", "scaffold_templates", "doctor",
)


def _check(name: str, ok: bool, **details: Any) -> dict[str, Any]:
    return {"name": name, "ok": ok, "details": details}


def _actual_files(skill_root: Path) -> set[str]:
    result = set()
    for path in skill_root.rglob("*"):
        if not path.is_file():
            continue
        relative = path.relative_to(skill_root)
        if relative == Path("header.md") or relative == Path("manifest.json"):
            continue
        if any(part in HOUSEKEEPING for part in relative.parts):
            continue
        if relative.suffix in {".pyc", ".pyo"}:
            continue
        result.add(relative.as_posix())
    return result


def _runtime(home: Path) -> tuple[list[str] | None, dict[str, Any]]:
    runtime_file = home / ".obsidian-kb-skill" / "runtime.json"
    try:
        payload = json.loads(runtime_file.read_text(encoding="utf-8"))
        command = payload["python"]
        valid = (
            payload.get("schema_version") == 1
            and isinstance(command, list)
            and bool(command)
            and all(isinstance(part, str) and part for part in command)
            and Path(command[0]).is_file()
        )
    except (OSError, KeyError, TypeError, json.JSONDecodeError) as exc:
        return None, _check("runtime", False, error=str(exc))
    if not valid:
        return None, _check("runtime", False, error="invalid runtime command")
    try:
        probe = subprocess.run(
            [*command, "-c", "import sys; print('.'.join(map(str, sys.version_info[:3])))"],
            capture_output=True,
            text=True,
            check=False,
        )
    except OSError as exc:
        return None, _check("runtime", False, command=command, error=str(exc))
    try:
        parsed_version = tuple(
            int(part) for part in probe.stdout.strip().split(".")[:2]
        )
    except ValueError:
        parsed_version = ()
    ok = probe.returncode == 0 and parsed_version >= (3, 11)
    return command if ok else None, _check(
        "runtime", ok, command=command, version=probe.stdout.strip(), error=probe.stderr.strip()
    )
```

Implement `inspect_installation` so malformed manifest returns five stable check
slots rather than raising; valid manifests compare file sets and hashes, runtime
is probed, dependencies import in the selected environment, and resources are
checked:

```python
def inspect_installation(skill_root: Path, home: Path | None = None) -> dict[str, Any]:
    root = skill_root.expanduser().resolve()
    resolved_home = (home or Path.home()).expanduser().resolve()
    checks: list[dict[str, Any]] = []
    manifest_file = root / "manifest.json"
    manifest: dict[str, Any] = {}
    manifest_error = "unsupported manifest"
    try:
        manifest = json.loads(manifest_file.read_text(encoding="utf-8"))
        files = manifest["files"]
        manifest_ok = (
            manifest.get("schema_version") == 1
            and manifest.get("product") == "obsidian-kb-skill"
            and isinstance(manifest.get("version"), str)
            and isinstance(files, dict)
            and all(isinstance(k, str) and isinstance(v, str) for k, v in files.items())
        )
    except (OSError, KeyError, TypeError, json.JSONDecodeError) as exc:
        manifest_ok = False
        files = {}
        manifest_error = str(exc)
    checks.append(_check("manifest", manifest_ok, error=None if manifest_ok else manifest_error))

    actual = _actual_files(root)
    expected = set(files)
    changed = sorted(
        relative for relative in actual & expected
        if hashlib.sha256((root / relative).read_bytes()).hexdigest() != files[relative]
    )
    checks.append(_check(
        "payload",
        manifest_ok and not (expected - actual) and not (actual - expected) and not changed,
        missing=sorted(expected - actual), extra=sorted(actual - expected), changed=changed,
    ))

    command, runtime_check = _runtime(resolved_home)
    checks.append(runtime_check)
    scripts = root / "scripts"
    vendor = resolved_home / ".obsidian-kb-skill" / "vendor"
    env = os.environ.copy()
    env["PYTHONPATH"] = os.pathsep.join([str(scripts), str(vendor)])
    code = "import yaml;" + ";".join(
        f"import obsidian_kb_skill.scripts.{name}" for name in HELPER_MODULES
    )
    dependency_probe = subprocess.run(
        [*(command or [sys.executable]), "-c", code],
        env=env, capture_output=True, text=True, check=False,
    )
    checks.append(_check(
        "dependencies", command is not None and dependency_probe.returncode == 0,
        error=dependency_probe.stderr.strip(),
    ))
    missing_resources = [name for name in REQUIRED_RESOURCES if not (root / name).is_file()]
    checks.append(_check("resources", not missing_resources, missing=missing_resources))
    return {
        "schema_version": "1.0",
        "ok": all(check["ok"] for check in checks),
        "version": manifest.get("version"),
        "skill_root": str(root),
        "checks": checks,
    }
```

```python
def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Diagnose an installed Obsidian Knowledge Base Skill.")
    parser.add_argument("--skill-root", type=Path)
    parser.add_argument("--json", action="store_true")
    args = parser.parse_args(argv)

    if args.json:

    else:
        print(f"Obsidian Knowledge Base Skill {result['version'] or 'unknown'}")
        for check in result["checks"]:
            print(f"{'PASS' if check['ok'] else 'FAIL'}  {check['name']}")
    return 0 if result["ok"] else 1


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
def _top_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        description="Run a helper bundled with the Obsidian Knowledge Base Skill."
    )
    parser.add_argument("helper", nargs="?", choices=sorted(HELPERS))
    return parser


def parse_dispatch(argv: list[str] | None = None) -> tuple[str, list[str]]:
    values = list(sys.argv[1:] if argv is None else argv)
    parser = _top_parser()
    if not values or values[0] in {"-h", "--help"}:
        parser.parse_args(values)
        raise AssertionError("argparse help exits")
    helper = values[0]
    if helper not in HELPERS:
        parser.error(
            f"argument helper: invalid choice: {helper!r} "
            f"(choose from {', '.join(repr(name) for name in sorted(HELPERS))})"
        )
    forwarded = values[1:]
    if forwarded[:1] == ["--"]:
        forwarded = forwarded[1:]
    return helper, forwarded


def main(argv: list[str] | None = None) -> int:
    helper, forwarded = parse_dispatch(argv)
    skill_root = Path(__file__).resolve().parent.parent
    command = [sys.executable] if helper == "doctor" else None
    if command is None:
        try:
            command = python_command()
        except RuntimeError as exc:
            print(f"error: {exc}", file=sys.stderr)
            return 3
    try:
        return subprocess.run(
            [*command, "-m", HELPERS[helper], *forwarded],
            env=helper_environment(skill_root), check=False,
        ).returncode
    except OSError as exc:
        print(f"error: helper runtime failed: {exc}", file=sys.stderr)
        return 3
```
